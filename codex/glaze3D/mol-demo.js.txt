const Q_MAX = 25.0,
  Q_MIN = 0.01;
const FF_N = 128;
const PAT_SIZE = 2048;
const MAX_ATOMS_JS = isMobile ? 16 : 60; // must match #define MAX_ATOMS in shader
const FF_TBL = {};
for (const [el, c] of Object.entries(FF)) {
  const t = new Float32Array(FF_N);
  for (let i = 0; i < FF_N; i++) {
    const q = Q_MIN + ((Q_MAX - Q_MIN) * i) / (FF_N - 1);
    const s = (q / (4 * Math.PI)) ** 2;
    t[i] =
      c[0] * Math.exp(-c[4] * s) +
      c[1] * Math.exp(-c[5] * s) +
      c[2] * Math.exp(-c[6] * s) +
      c[3] * Math.exp(-c[7] * s) +
      c[8];
  }
  FF_TBL[el] = t;
}
function ff(el, q) {
  const i = Math.max(
    0,
    Math.min(FF_N - 1, Math.round(((q - Q_MIN) / (Q_MAX - Q_MIN)) * (FF_N - 1))),
  );
  return FF_TBL[el][i];
}

// ─── X-ray colormap LUT (256 entries, γ=0.72 pre-applied, built once) ────────
const XRAY_LUT = (() => {
  const stops = [
    [0.0, [3, 4, 18]],
    [0.18, [6, 18, 55]],
    [0.35, [4, 55, 105]],
    [0.5, [0, 105, 155]],
    [0.63, [0, 158, 188]],
    [0.74, [0, 198, 218]],
    [0.84, [45, 228, 235]],
    [0.92, [120, 242, 250]],
    [0.97, [195, 250, 255]],
    [1.0, [255, 255, 255]],
  ];
  const lut = new Uint8Array(256 * 3);
  for (let i = 0; i < 256; i++) {
    const v = (i / 255) ** 0.72;
    for (let s = 0; s < stops.length - 1; s++) {
      const [t0, c0] = stops[s],
        [t1, c1] = stops[s + 1];
      if (v <= t1) {
        const t = (v - t0) / (t1 - t0);
        lut[i * 3] = Math.round(c0[0] + t * (c1[0] - c0[0]));
        lut[i * 3 + 1] = Math.round(c0[1] + t * (c1[1] - c0[1]));
        lut[i * 3 + 2] = Math.round(c0[2] + t * (c1[2] - c0[2]));
        break;
      }
    }
  }
  return lut;
})();

// ─── WebGL2 scatter shader ────────────────────────────────────────────────────
const VS_SRC = `#version 300 es
void main() {
    const vec2 pos[4] = vec2[4](
        vec2(-1,-1), vec2(1,-1), vec2(-1,1), vec2(1,1)
    );
    gl_Position = vec4(pos[gl_VertexID], 0.0, 1.0);
}`;

// Fragment shader computes IAM: I(qx,qy) = q * |sum_i f_i(q) exp(i*q.r_i)|^2
// Atom positions are projected into the detector plane by the caller.
// Form factors: f(q) = dot(a, exp(-b*s)) + c, s=(q/4pi)^2.
// Colormap passed as 256x1 RGB texture.
const FS_SRC = `#version 300 es
precision highp float;
#define MAX_ATOMS ${MAX_ATOMS_JS}
#define PI 3.141592653589793

uniform vec2  uResolution;
uniform float uQmin;
uniform float uQmax;
uniform int   uNatoms;
uniform vec2  uAtomPos[MAX_ATOMS];
uniform vec4  uFFa[MAX_ATOMS];
uniform vec4  uFFb[MAX_ATOMS];
uniform float uFFc[MAX_ATOMS];
uniform float uNorm;
uniform sampler2D uLUT;

out vec4 fragColor;

void main() {
    vec2  center = uResolution * 0.5;
    float R      = min(center.x, center.y) * 0.97;
    float sc     = uQmax / R;
    // gl_FragCoord.y is 0 at screen-bottom; canvas py=0 is screen-top.
    // CPU convention: qy = (cy - py)*sc, positive at canvas top.
    // Equivalent here: qy = (fragCoord.y - center.y)*sc (both increase same direction).
    float qx     = (gl_FragCoord.x - center.x) * sc;
    float qy     = (gl_FragCoord.y - center.y) * sc;
    float q      = length(vec2(qx, qy));

    if (q < uQmin || q > uQmax) { fragColor = vec4(0.0); return; }

    float s = q / (4.0 * PI);
    s *= s;

    float re = 0.0, im = 0.0;
    for (int i = 0; i < MAX_ATOMS; i++) {
        if (i >= uNatoms) break;
        float f  = dot(uFFa[i], exp(-uFFb[i] * s)) + uFFc[i];
        float ph = qx * uAtomPos[i].x + qy * uAtomPos[i].y;
        re += f * cos(ph);
        im += f * sin(ph);
    }

    float I_raw = re*re + im*im;
    float fade  = min(1.0, (q - uQmin) / (uQmin * 2.0))
                * min(1.0, (uQmax - q) / (uQmax * 0.06));
    float v     = clamp(log(1.0 + I_raw) / uNorm, 0.0, 1.0);
    vec3  col   = texture(uLUT, vec2(v, 0.5)).rgb;

    fragColor = vec4(col, v * fade);
}`;

function makeShader(gl, type, src) {
  const sh = gl.createShader(type);
  gl.shaderSource(sh, src);
  gl.compileShader(sh);
  if (!gl.getShaderParameter(sh, gl.COMPILE_STATUS))
    throw new Error("Shader: " + gl.getShaderInfoLog(sh));
  return sh;
}

function makeProgram(gl, vsSrc, fsSrc) {
  const prog = gl.createProgram();
  gl.attachShader(prog, makeShader(gl, gl.VERTEX_SHADER, vsSrc));
  gl.attachShader(prog, makeShader(gl, gl.FRAGMENT_SHADER, fsSrc));
  gl.linkProgram(prog);
  if (!gl.getProgramParameter(prog, gl.LINK_STATUS))
    throw new Error("Program: " + gl.getProgramInfoLog(prog));
  return prog;
}

let gl = null,
  glProg = null,
  glUniforms = {};
let gpuPath = false;

function setAtomFFUniforms(mol) {
  gl.uniform1i(glUniforms["uNatoms"], mol.atoms.length);
  for (let i = 0; i < mol.atoms.length; i++) {
    const c = FF[mol.atoms[i]];
    gl.uniform4f(glUniforms[`uFFa${i}`], c[0], c[1], c[2], c[3]);
    gl.uniform4f(glUniforms[`uFFb${i}`], c[4], c[5], c[6], c[7]);
    gl.uniform1f(glUniforms[`uFFc${i}`], c[8]);
  }
}

function initGPU() {
  gl = patCanvas.getContext("webgl2", { alpha: true, premultipliedAlpha: false });
  if (!gl) return;
  try {
    glProg = makeProgram(gl, VS_SRC, FS_SRC);
    gl.useProgram(glProg);

    const names = ["uResolution", "uQmin", "uQmax", "uNatoms", "uNorm", "uLUT"];
    for (const n of names) glUniforms[n] = gl.getUniformLocation(glProg, n);
    for (let i = 0; i < MAX_ATOMS_JS; i++) {
      glUniforms[`uAtomPos${i}`] = gl.getUniformLocation(glProg, `uAtomPos[${i}]`);
      glUniforms[`uFFa${i}`] = gl.getUniformLocation(glProg, `uFFa[${i}]`);
      glUniforms[`uFFb${i}`] = gl.getUniformLocation(glProg, `uFFb[${i}]`);
      glUniforms[`uFFc${i}`] = gl.getUniformLocation(glProg, `uFFc[${i}]`);
    }

    const lutTex = gl.createTexture();
    gl.activeTexture(gl.TEXTURE0);
    gl.bindTexture(gl.TEXTURE_2D, lutTex);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGB, 256, 1, 0, gl.RGB, gl.UNSIGNED_BYTE, XRAY_LUT);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
    gl.uniform1i(glUniforms["uLUT"], 0);

    gl.uniform1f(glUniforms["uQmin"], Q_MIN);
    gl.uniform1f(glUniforms["uQmax"], Q_MAX);

    setAtomFFUniforms(MOL);
    gpuPath = true;
  } catch (e) {
    console.warn("WebGL2 scatter init failed, using CPU fallback:", e);
    gl = null;
  }
}

// ─── Pattern computation ──────────────────────────────────────────────────────
// Physics: I(qx,qy) = |Σᵢ fᵢ(|q|)·exp(i·(qx·xᵢ'+qy·yᵢ'))|²
// (xᵢ',yᵢ') = atom positions projected into camera plane (qz = 0 on detector)
// Weighted by q·I to moderately suppress the forward-scattering envelope.

const PAT_MS = 33; // ~30fps for pattern updates
let lastPatTs = 0;
let prevCamQuat = new THREE.Quaternion();
let patDirty = true;

// Normalization via 32x32 CPU grid: same 97th-percentile algorithm as the
// original but ~90x fewer samples, fast enough to run each frame.
// EMA smoothing prevents per-frame jitter from the coarse grid.
let smoothNorm = 0;
function computeNorm32(atoms, W, H) {
  const cx = W / 2,
    cy = H / 2;
  const R = Math.min(cx, cy) * 0.97;
  const sc = Q_MAX / R;
  const sx = W / 32,
    sy = H / 32;
  const buf = new Float32Array(32 * 32);
  let k = 0;
  for (let row = 0; row < 32; row++) {
    for (let col = 0; col < 32; col++) {
      const px = (col + 0.5) * sx,
        py = (row + 0.5) * sy;
      const qx = (px - cx) * sc,
        qy = (cy - py) * sc;
      const q = Math.sqrt(qx * qx + qy * qy);
      if (q < Q_MIN || q > Q_MAX) continue;
      const s = (q / (4 * Math.PI)) ** 2;
      let re = 0,
        im = 0;
      for (const { el, vx, vy } of atoms) {
        const f = ff(el, q);
        const ph = qx * vx + qy * vy;
        re += f * Math.cos(ph);
        im += f * Math.sin(ph);
      }
      buf[k++] = Math.log(1 + re * re + im * im);
    }
  }
  if (!k) return smoothNorm || 1;
  const sub = buf.subarray(0, k);
  sub.sort();
  const raw = sub[Math.floor(k * 0.97)] || 1;
  smoothNorm = smoothNorm > 0 ? smoothNorm + 0.25 * (raw - smoothNorm) : raw;
  return smoothNorm;
}

function drawPattern() {
  const now = performance.now();
  if (now - lastPatTs < PAT_MS) return;
  if (!patDirty && camera.quaternion.angleTo(prevCamQuat) < 0.003) return;
  prevCamQuat.copy(camera.quaternion);
  patDirty = false;
  lastPatTs = now;

  const W = patCanvas.width,
    H = patCanvas.height;
  if (!W || !H) return;

  camera.updateMatrixWorld();
  const me = camera.matrixWorldInverse.elements;
  const atoms = MOL.atoms.map((el, i) => {
    const [x, y, z] = MOL.pos[i];
    return { el, vx: me[0] * x + me[4] * y + me[8] * z, vy: me[1] * x + me[5] * y + me[9] * z };
  });

  if (gpuPath) {
    const norm = computeNorm32(atoms, W, H);
    gl.useProgram(glProg);
    gl.viewport(0, 0, W, H);
    gl.clearColor(0, 0, 0, 0);
    gl.clear(gl.COLOR_BUFFER_BIT);
    gl.uniform2f(glUniforms["uResolution"], W, H);
    gl.uniform1f(glUniforms["uNorm"], norm);
    for (let i = 0; i < atoms.length; i++)
      gl.uniform2f(glUniforms[`uAtomPos${i}`], atoms[i].vx, atoms[i].vy);
    gl.drawArrays(gl.TRIANGLE_STRIP, 0, 4);
    return;
  }

  // ── CPU fallback ──────────────────────────────────────────────────────────
  const cx = W / 2,
    cy = H / 2;
  const R = Math.min(cx, cy) * 0.97;
  const sc = Q_MAX / R;

  raw.fill(0);
  mask.fill(0);
  let maxV = 0;

  for (let py = 0; py < H; py++) {
    for (let px = 0; px < W; px++) {
      const qx = (px - cx) * sc,
        qy = (cy - py) * sc;
      const q = Math.sqrt(qx * qx + qy * qy);
      if (q < Q_MIN || q > Q_MAX) continue;
      let re = 0,
        im = 0;
      for (const { el, vx, vy } of atoms) {
        const f = ff(el, q);
        const ph = qx * vx + qy * vy;
        re += f * Math.cos(ph);
        im += f * Math.sin(ph);
      }
      const v = Math.log(1 + re * re + im * im);
      const idx = py * W + px;
      raw[idx] = v;
      mask[idx] = 1;
      fadeTbl[idx] =
        Math.min(1, (q - Q_MIN) / (Q_MIN * 2)) * Math.min(1, (Q_MAX - q) / (Q_MAX * 0.06));
      if (v > maxV) maxV = v;
    }
  }
  if (maxV === 0) return;

  let sampLen = 0;
  for (let i = 0; i < raw.length; i += 8) if (raw[i] > 0) sampBuf[sampLen++] = raw[i];
  const samp = sampBuf.subarray(0, sampLen);
  samp.sort();
  const norm = samp[Math.floor(sampLen * 0.97)] || maxV;

  const imgData = patCtx.createImageData(W, H);
  const px4 = imgData.data;
  for (let i = 0; i < W * H; i++) {
    const base = i * 4;
    if (!mask[i]) {
      px4[base + 3] = 0;
      continue;
    }
    const vi = Math.min(255, ((raw[i] / norm) * 255) | 0);
    const li = vi * 3;
    px4[base] = XRAY_LUT[li];
    px4[base + 1] = XRAY_LUT[li + 1];
    px4[base + 2] = XRAY_LUT[li + 2];
    px4[base + 3] = (vi * fadeTbl[i]) | 0;
  }
  patCtx.putImageData(imgData, 0, 0);
}

// ─── .mol / .sdf V2000 parser ────────────────────────────────────────────────
// SDF files are one or more mol records separated by $$$$; we parse the first.
// The mol block is always at the top, so reading by atom/bond count stops
// before any SDF data fields naturally.
function parseMol(text, filename) {
  const lines = text.replace(/\r\n/g, "\n").split("\n");
  const rawName = lines[0].trim();
  // Fall back to filename (minus extension) if the name field is blank or numeric
  const name =
    rawName && !/^\d+$/.test(rawName)
      ? rawName
      : (filename || "").replace(/\.(mol|sdf)$/i, "") || "Unknown";

  // Counts line contains "V2000"; some files omit the comment line so it lands at [2] not [3]
  let ci = 3;
  for (let i = 2; i <= 4; i++) {
    if ((lines[i] || "").includes("V2000")) {
      ci = i;
      break;
    }
  }

  const counts = lines[ci] || "";
  const natoms = parseInt(counts.substring(0, 3), 10);
  const nbonds = parseInt(counts.substring(3, 6), 10);
  if (!natoms || isNaN(natoms)) throw new Error("Could not parse atom count");

  const atoms = [],
    pos = [],
    bonds = [];
  for (let i = 0; i < natoms; i++) {
    const parts = (lines[ci + 1 + i] || "").trim().split(/\s+/);
    pos.push([parseFloat(parts[0]), parseFloat(parts[1]), parseFloat(parts[2])]);
    atoms.push(parts[3]);
  }
  for (let i = 0; i < nbonds; i++) {
    const parts = (lines[ci + 1 + natoms + i] || "").trim().split(/\s+/);
    bonds.push([parseInt(parts[0], 10) - 1, parseInt(parts[1], 10) - 1]);
  }

  // Center on centroid so the molecule rotates about its own center
  const cx = pos.reduce((s, p) => s + p[0], 0) / pos.length;
  const cy = pos.reduce((s, p) => s + p[1], 0) / pos.length;
  const cz = pos.reduce((s, p) => s + p[2], 0) / pos.length;
  pos.forEach((p) => {
    p[0] -= cx;
    p[1] -= cy;
    p[2] -= cz;
  });

  return { name, atoms, pos, bonds };
}

// ─── Dynamic molecule switching ───────────────────────────────────────────────
function switchMolecule(mol) {
  const unknown = [...new Set(mol.atoms.filter((el) => !(el in FF)))];
  if (unknown.length) {
    console.warn("mol-demo: unknown elements (no form factors):", unknown.join(", "));
    return;
  }
  if (mol.atoms.length > MAX_ATOMS_JS) {
    console.warn(`mol-demo: molecule has ${mol.atoms.length} atoms, max is ${MAX_ATOMS_JS}`);
    return;
  }
  MOL = mol;
  smoothNorm = 0;
  buildMolMesh(mol);
  if (gpuPath) {
    gl.useProgram(glProg);
    setAtomFFUniforms(mol);
  }
  patDirty = true;
}

// ─── Resize handlers ──────────────────────────────────────────────────────────
function resizeMol() {
  const rect = molCanvas.getBoundingClientRect();
  const w = Math.round(rect.width),
    h = Math.round(rect.height);
  if (!w || !h) return;
  renderer.setSize(w, h, false);
  camera.aspect = w / h;
  camera.updateProjectionMatrix();
}

function resizePat() {
  const BUF = isMobile ? 600 : 900;
  patCanvas.width = BUF;
  patCanvas.height = BUF;
  if (gpuPath) {
    gl.viewport(0, 0, BUF, BUF);
  } else {
    if (cpuCanvas) {
      cpuCanvas.width = BUF;
      cpuCanvas.height = BUF;
    }
    const N = BUF * BUF;
    raw = new Float32Array(N);
    mask = new Uint8Array(N);
    fadeTbl = new Float32Array(N);
    sampBuf = new Float32Array(Math.ceil(N / 8));
  }
  patDirty = true;
}

// ─── Three.js scene ───────────────────────────────────────────────────────────
const renderer = new THREE.WebGLRenderer({ canvas: molCanvas, antialias: true, alpha: true });
renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
renderer.setClearColor(0x000000, 0);

const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(40, 1, 0.01, 200);
camera.position.set(4, 3, 11);
scene.add(camera);

const controls = new OrbitControls(camera, renderer.domElement);
controls.enablePan = false;
controls.enableZoom = false;
controls.autoRotate = true;
controls.autoRotateSpeed = 0.75;
controls.enableDamping = true;
controls.dampingFactor = 0.07;

scene.add(new THREE.AmbientLight(0xffffff, 0.5));
const keyLight = new THREE.DirectionalLight(0xffffff, 1.2);
keyLight.position.set(5, 8, -6);
camera.add(keyLight);
const fillLight = new THREE.DirectionalLight(0x88aaff, 0.4);
fillLight.position.set(-4, -2, -4);
camera.add(fillLight);
const rimLight = new THREE.DirectionalLight(0x44ccff, 0.25);
rimLight.position.set(0, -5, 5);
camera.add(rimLight);

// ─── Molecule mesh ────────────────────────────────────────────────────────────
const molGroup = new THREE.Group();
scene.add(molGroup);

function buildMolMesh(mol) {
  molGroup.clear();
  mol.atoms.forEach((el, i) => {
    const { r, col } = ATOM[el] || ATOM_FALLBACK;
    const geo = new THREE.SphereGeometry(r * 0.45, 32, 20);
    const mat = new THREE.MeshPhongMaterial({ color: col, shininess: 100, specular: 0x333333 });
    const mesh = new THREE.Mesh(geo, mat);
    mesh.position.fromArray(mol.pos[i]);
    molGroup.add(mesh);
  });
  mol.bonds.forEach(([a, b]) => {
    const p1 = new THREE.Vector3().fromArray(mol.pos[a]);
    const p2 = new THREE.Vector3().fromArray(mol.pos[b]);
    const dir = new THREE.Vector3().subVectors(p2, p1);
    const len = dir.length();
    const mid = p1.clone().lerp(p2, 0.5);
    const geo = new THREE.CylinderGeometry(0.065, 0.065, len, 16);
    const mat = new THREE.MeshPhongMaterial({ color: 0x888888, shininess: 70 });
    const mesh = new THREE.Mesh(geo, mat);
    mesh.position.copy(mid);
    mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir.normalize());
    molGroup.add(mesh);
  });
  if (molNameEl) molNameEl.textContent = mol.name;
}

// ─── Animation loop ───────────────────────────────────────────────────────────
let prevTs = 0;
let rafId = null;

function loop(ts) {
  prevTs = ts;
  rafId = requestAnimationFrame(loop);
  try {
    controls.update();
    renderer.render(scene, camera);
  } catch (_) {
    /* WebGL context may be temporarily lost */
  }
  drawPattern();
}

// ─── Init ─────────────────────────────────────────────────────────────────────
requestAnimationFrame(() => {
  initGPU();
  if (!gpuPath) initCPU();

  const pool = MOLECULES.filter((m) => m.atoms <= MAX_ATOMS_JS);
  const randomSdf = pool[Math.floor(Math.random() * pool.length)].file;
  resizeMol();
  resizePat();
  window.addEventListener("resize", () => {
    resizeMol();
    resizePat();
  });
  controls.addEventListener("change", () => {
    patDirty = true;
  });

  // Drag-and-drop a .mol file onto the molecule area to switch molecule.
  // patCanvas has pointer-events:none in CSS, so listeners go on the parent
  // (.mol-stage), which bubbles events from molCanvas and the label.
  const dropZone = patCanvas.parentElement;
  dropZone.addEventListener("dragover", (e) => e.preventDefault());
  dropZone.addEventListener("drop", (e) => {
    e.preventDefault();
    const file = e.dataTransfer.files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (evt) => {
      try {
        switchMolecule(parseMol(evt.target.result, file.name));
      } catch (err) {
        console.error("mol-demo: failed to parse .mol file:", err);
      }
    };
    reader.readAsText(file);
  });

  const startLoop = () => {
    rafId = requestAnimationFrame((ts) => {
      prevTs = ts;
      loop(ts);
    });
  };
  fetch("/static/molecules/" + encodeURIComponent(randomSdf))
    .then((r) => r.text())
    .then((text) => {
      try {
        switchMolecule(parseMol(text, randomSdf));
      } catch (_) {
        buildMolMesh(MOL);
      }
    })
    .catch(() => {
      buildMolMesh(MOL);
    })
    .finally(startLoop);
});

// ─── Page lifecycle ───────────────────────────────────────────────────────────
window.addEventListener("pageshow", (e) => {
  patDirty = true;
  if (e.persisted) {
    resizePat();
    resizeMol();
    if (!rafId)
      rafId = requestAnimationFrame((ts) => {
        prevTs = ts;
        loop(ts);
      });
  }
});

document.addEventListener("visibilitychange", () => {
  if (!document.hidden) patDirty = true;
});
